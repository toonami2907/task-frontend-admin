import React, { useState } from "react";
import PropTypes from "prop-types";
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogFooter,
} from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { Plus, Search, X } from "lucide-react";
import api from "@/functions/resources/axiosConfig";
import { TimePicker } from "./TimePicker";
import { ErrorToast } from "@/hooks/customToast";
import toast from "react-hot-toast";
import { conditionConfig } from "@/lib/rule.config";
import { cn } from "@/lib/utils";

export const CreateRuleForm = ({ isOpen, onClose, onSubmit }) => {
  const [showTimeRange, setShowTimeRange] = useState(false);
  const [activeTargetLevel, setActiveTargetLevel] =
    useState("campaign");
  const [searchTerm, setSearchTerm] = useState("");
  const [searchResults, setSearchResults] = useState([]);
  const [loading, setLoading] = useState(false);

  const [formData, setFormData] = useState({
    name: "",
    targetId: [],
    conditions: [],
    actions: [],
    frequency: "",
    frequencyValue: 1,
    targetLevel: activeTargetLevel,
    timeRanges: [],
    status: "active",
  });
  const targetLevelTabs = [
    { id: "campaign", label: "Campaigns" },
    { id: "adSet", label: "Ad Sets" },
    { id: "ad", label: "Ads" },
  ];
  const handleSubmit = async (e) => {
    e.preventDefault();
    console.log(formData);
    try {
      await onSubmit(formData);
      setFormData({
        name: "",
        targetId: [],
        conditions: [],
        actions: [],
        frequency: "",
        frequencyValue: 1,
        targetLevel: "",
        timeRanges: [],
        status: "active",
      });
      toast.success("Rule Created sucessfully");
      onClose();
    } catch (error) {
      console.error("error", error?.message);
      return ErrorToast(
        error?.message || "error while creating a rule "
      );
    }
  };

  const handleTargetSearch = async (searchTerm) => {
    setSearchTerm(searchTerm);
    if (searchTerm.length < 2) {
      setSearchResults([]);
      return;
    }

    setLoading(true);
    try {
      const [campaigns, adSets, ads] = await Promise.all([
        api.get("/auth/db/campaigns"),
        api.get("/auth/db/ads_set"),
        api.get("/auth/db/ads"),
      ]);

      const searchValue = searchTerm.toLowerCase();

      const results = [
        ...campaigns.data.map((c) => ({
          id: c.campaignId,
          name: c.campaignName,
          level: "campaign",
        })),
        ...adSets.data.map((a) => ({
          id: a.adSetId,
          name: a.adSetName,
          level: "adSet",
        })),
        ...ads.data.map((a) => ({
          id: a.adId,
          name: a.adName,
          level: "ad",
        })),
      ].filter((item) =>
        item.name.toLowerCase().includes(searchValue)
      );

      setSearchResults(results);
    } catch (error) {
      console.error("Search error:", error);
      setSearchResults([]);
    } finally {
      setLoading(false);
    }
  };

  const addCondition = () => {
    const now = new Date();
    const start = new Date(now.setHours(9, 0, 0));
    const end = new Date(now.setHours(17, 0, 0));

    setFormData((prev) => ({
      ...prev,
      conditions: [
        ...prev.conditions,
        {
          id: crypto.randomUUID(),
          metric: "",
          operator: "",
          value: "",
          timeframe: "",
          timeRanges: [{ start, end }], // Initialize with default time range
          logicalOperator: prev.conditions.length > 0 ? "AND" : "",
        },
      ],
    }));
  };
  const updateCondition = (id, field, value) => {
    setFormData((prev) => ({
      ...prev,
      conditions: prev.conditions.map((condition) =>
        condition.id === id
          ? { ...condition, [field]: value }
          : condition
      ),
    }));
  };

  const removeCondition = (id) => {
    setFormData((prev) => ({
      ...prev,
      conditions: prev.conditions.filter(
        (condition) => condition.id !== id
      ),
    }));
  };

  const addAction = () => {
    setFormData((prev) => ({
      ...prev,
      actions: [
        ...prev.actions,
        {
          id: crypto.randomUUID(),
          type: "",
          value: "",
          customValue: "",
        },
      ],
    }));
  };

  const updateAction = (id, field, value) => {
    setFormData((prev) => ({
      ...prev,
      actions: prev.actions.map((action) =>
        action.id === id ? { ...action, [field]: value } : action
      ),
    }));
  };

  const removeAction = (id) => {
    setFormData((prev) => ({
      ...prev,
      actions: prev.actions.filter((action) => action.id !== id),
    }));
  };
  const addTimeRange = () => {
    const now = new Date();
    const start = new Date(now.setHours(9, 0, 0));
    const end = new Date(now.setHours(17, 0, 0));

    setFormData((prev) => ({
      ...prev,
      timeRanges: [
        ...prev.timeRanges,
        {
          id: crypto.randomUUID(),
          startTime: start,
          endTime: end,
        },
      ],
    }));
  };

  const updateTimeRange = (id, field, value) => {
    setFormData((prev) => ({
      ...prev,
      timeRanges: prev.timeRanges.map((range) =>
        range.id === id ? { ...range, [field]: value } : range
      ),
    }));
  };

  const removeTimeRange = (id) => {
    setFormData((prev) => ({
      ...prev,
      timeRanges: prev.timeRanges.filter((range) => range.id !== id),
    }));
  };

  const getActionOptions = (actionType) => {
    switch (actionType) {
      case "change_budget":
        return (
          <>
            <SelectItem value="increase_percent">
              Increase by %
            </SelectItem>
            <SelectItem value="decrease_percent">
              Decrease by %
            </SelectItem>
            <SelectItem value="increase_fixed">
              Increase by fixed amount
            </SelectItem>
            <SelectItem value="decrease_fixed">
              Decrease by fixed amount
            </SelectItem>
            <SelectItem value="set_fixed">
              Set to fixed amount
            </SelectItem>
          </>
        );
      case "change_status":
        return (
          <>
            <SelectItem value="active">Set Active</SelectItem>
            <SelectItem value="paused">Set Paused</SelectItem>
          </>
        );
      case "change_name":
        return (
          <>
            <SelectItem value="prefix">Add prefix</SelectItem>
            <SelectItem value="suffix">Add suffix</SelectItem>
          </>
        );
      default:
        return null;
    }
  };
  const renderConditionValue = (condition) => {
    const metric = conditionConfig.metrics[condition.metric];
    if (!metric) return null;

    switch (metric.valueType) {
      case "number":
        return (
          <Input
            type="number"
            value={condition.value}
            onChange={(e) =>
              updateCondition(condition.id, "value", e.target.value)
            }
            placeholder="Value"
            className="w-1/4"
          />
        );

      case "timeRange":
        return (
          <div className="flex gap-2 items-center">
            <TimePicker
              value={condition.timeRanges[0].start}
              onChange={(value) =>
                updateCondition(condition.id, "timeRanges", [
                  { ...condition.timeRanges[0], start: value },
                ])
              }
              label="Start"
            />
            <span>to</span>
            <TimePicker
              value={condition.timeRanges[0].end}
              onChange={(value) =>
                updateCondition(condition.id, "timeRanges", [
                  { ...condition.timeRanges[0], end: value },
                ])
              }
              label="End"
            />
          </div>
        );

      default:
        return (
          <Input
            type="text"
            value={condition.value}
            onChange={(e) =>
              updateCondition(condition.id, "value", e.target.value)
            }
            placeholder="Value"
            className="w-1/4"
          />
        );
    }
  };
  const renderConditions = () => (
    <div>
      <div className="flex justify-between items-center mb-2">
        <label className="block text-sm font-medium">
          Conditions
        </label>
        <Button
          type="button"
          variant="outline"
          size="sm"
          onClick={addCondition}
        >
          <Plus className="h-4 w-4 mr-2" />
          Add Condition
        </Button>
      </div>
      {formData.conditions.map((condition, index) => (
        <div key={condition.id} className="space-y-2 mb-4">
          {index > 0 && (
            <Select
              value={condition.logicalOperator}
              onValueChange={(value) =>
                updateCondition(
                  condition.id,
                  "logicalOperator",
                  value
                )
              }
            >
              <SelectTrigger className="w-20">
                <SelectValue />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="AND">AND</SelectItem>
                <SelectItem value="OR">OR</SelectItem>
              </SelectContent>
            </Select>
          )}
          <div className="flex gap-2 items-center">
            <Select
              value={condition.metric}
              onValueChange={(value) =>
                updateCondition(condition.id, "metric", value)
              }
            >
              <SelectTrigger className="w-1/3">
                <SelectValue placeholder="Select metric" />
              </SelectTrigger>
              <SelectContent>
                {Object.entries(conditionConfig.metrics).map(
                  ([key, config]) => (
                    <SelectItem key={key} value={key}>
                      {config.label}
                    </SelectItem>
                  )
                )}
              </SelectContent>
            </Select>

            {condition.metric && (
              <Select
                value={condition.operator}
                onValueChange={(value) =>
                  updateCondition(condition.id, "operator", value)
                }
              >
                <SelectTrigger className="w-1/3">
                  <SelectValue placeholder="Select operator" />
                </SelectTrigger>
                <SelectContent>
                  {conditionConfig.metrics[
                    condition.metric
                  ].operators.map((op) => (
                    <SelectItem key={op} value={op}>
                      {conditionConfig.operators[op]}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            )}

            {condition.metric === "spend" && (
              <Select
                value={condition.timeframe}
                onValueChange={(value) =>
                  updateCondition(condition.id, "timeframe", value)
                }
              >
                <SelectTrigger className="w-1/3">
                  <SelectValue placeholder="Select timeframe" />
                </SelectTrigger>
                <SelectContent>
                  {conditionConfig.metrics.spend.timeframes.map(
                    (timeframe) => (
                      <SelectItem key={timeframe} value={timeframe}>
                        {timeframe.replace("_", " ").toLowerCase()}
                      </SelectItem>
                    )
                  )}
                </SelectContent>
              </Select>
            )}

            {condition.operator && renderConditionValue(condition)}

            <Button
              type="button"
              variant="ghost"
              size="sm"
              onClick={() => removeCondition(condition.id)}
            >
              <X className="h-4 w-4" />
            </Button>
          </div>
        </div>
      ))}
    </div>
  );

  const renderTargetSelection = () => (
    <div className="space-y-4">
      <div className="relative">
        <Input
          type="text"
          placeholder="Search campaigns, ad sets, or ads..."
          value={searchTerm}
          onChange={(e) => handleTargetSearch(e.target.value)}
          className="pr-10"
        />
        {loading ? (
          <div className="absolute right-3 top-1/2 transform -translate-y-1/2">
            <div className="animate-spin h-4 w-4 border-2 border-blue-500 rounded-full border-t-transparent" />
          </div>
        ) : (
          <Search className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400" />
        )}
      </div>

      {searchResults.length > 0 && (
        <div className="space-y-2 max-h-60 overflow-y-auto border rounded-md p-2">
          <div className="flex items-center gap-2 p-2 border-b">
            <input
              type="checkbox"
              id="select-all"
              checked={
                formData.targetId.length === searchResults.length
              }
              onChange={(e) => {
                const newTargetId = e.target.checked
                  ? searchResults.map((item) => item.id)
                  : [];
                setFormData((prev) => ({
                  ...prev,
                  targetId: newTargetId,
                }));
              }}
              className="rounded border-gray-300"
            />
            <label htmlFor="select-all" className="font-medium">
              Select All Results
            </label>
          </div>

          {searchResults.map((result) => (
            <div
              key={result.id}
              className="flex items-center gap-2 p-2 hover:bg-gray-50"
            >
              <input
                type="checkbox"
                id={result.id}
                checked={formData.targetId.includes(result.id)}
                onChange={() => {
                  setFormData((prev) => ({
                    ...prev,
                    targetId: prev.targetId.includes(result.id)
                      ? prev.targetId.filter(
                          (targetId) => targetId !== result.id
                        )
                      : [...prev.targetId, result.id],
                  }));
                }}
                className="rounded border-gray-300"
              />
              <label
                htmlFor={result.id}
                className="flex-1 cursor-pointer"
              >
                <div className="text-sm font-medium">
                  {result.name}{" "}
                  <span className="text-gray-500">
                    [{result.level}]
                  </span>
                </div>
                <div className="text-xs text-gray-500">
                  ID: {result.id}
                </div>
              </label>
            </div>
          ))}
        </div>
      )}
    </div>
  );

  // ... (rest of your existing code for conditions, actions, etc.)

  return (
    <Dialog open={isOpen} onOpenChange={onClose}>
      <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>Create New Rule</DialogTitle>
        </DialogHeader>
        <form onSubmit={handleSubmit} className="space-y-6">
          <div>
            <label className="block text-sm font-medium mb-1">
              Rule Name
            </label>
            <Input
              value={formData.name}
              onChange={(e) =>
                setFormData((prev) => ({
                  ...prev,
                  name: e.target.value,
                }))
              }
              placeholder="Enter rule name"
              required
            />
          </div>

          {renderTargetSelection()}

          {/* Your existing conditions, actions, frequency sections */}
          {renderConditions()}
          <div>
            <div className="flex justify-between items-center mb-2">
              <label className="block text-sm font-medium">
                Actions
              </label>
              <Button
                type="button"
                variant="outline"
                size="sm"
                onClick={addAction}
              >
                <Plus className="h-4 w-4 mr-2" />
                Add Action
              </Button>
            </div>
            {formData.actions.map((action) => (
              <div
                key={action.id}
                className="flex gap-2 mb-2 items-center"
              >
                <Select
                  value={action.type}
                  onValueChange={(value) =>
                    updateAction(action.id, "type", value)
                  }
                >
                  <SelectTrigger className="w-1/3">
                    <SelectValue placeholder="Select action" />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value="change_budget">
                      Change Budget
                    </SelectItem>
                    <SelectItem value="change_status">
                      Change Status
                    </SelectItem>
                    <SelectItem value="change_name">
                      Change Name
                    </SelectItem>
                    <SelectItem value="change_bid">
                      Change Bid
                    </SelectItem>
                    <SelectItem value="send_notification">
                      Send Notification
                    </SelectItem>
                  </SelectContent>
                </Select>

                <Select
                  value={action.value}
                  onValueChange={(value) =>
                    updateAction(action.id, "value", value)
                  }
                >
                  <SelectTrigger className="w-1/3">
                    <SelectValue placeholder="Select value" />
                  </SelectTrigger>
                  <SelectContent>
                    {getActionOptions(action.type)}
                  </SelectContent>
                </Select>

                {/* Custom input based on action type */}
                {(action.type === "change_budget" ||
                  action.type === "change_name") && (
                  <Input
                    type={
                      action.type === "change_budget"
                        ? "number"
                        : "text"
                    }
                    value={action.customValue || ""}
                    onChange={(e) =>
                      updateAction(
                        action.id,
                        "customValue",
                        e.target.value
                      )
                    }
                    placeholder={
                      action.type === "change_budget"
                        ? "Enter amount"
                        : "Enter text"
                    }
                    className="w-1/4"
                  />
                )}

                <Button
                  type="button"
                  variant="ghost"
                  size="sm"
                  onClick={() => removeAction(action.id)}
                >
                  <X className="h-4 w-4" />
                </Button>
              </div>
            ))}
          </div>
          {/* FREQUENCY SECTION */}
          <div className="flex gap-2 items-center">
            <Select
              value={formData.frequency}
              onValueChange={(value) => {
                setFormData((prev) => ({
                  ...prev,
                  frequency: value,
                }));
                setShowTimeRange(
                  ["hourly", "daily", "weekly"].includes(value)
                );
              }}
            >
              <SelectTrigger className="w-1/3">
                <SelectValue placeholder="Select frequency" />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="minutes">Minutes</SelectItem>
                <SelectItem value="hourly">Hourly</SelectItem>
                <SelectItem value="daily">Daily</SelectItem>
                <SelectItem value="weekly">Weekly</SelectItem>
              </SelectContent>
            </Select>

            <div className="flex items-center gap-2">
              <span className="text-sm font-medium">Every</span>
              <Input
                type="number"
                value={formData.frequencyValue}
                onChange={(e) =>
                  setFormData((prev) => ({
                    ...prev,
                    frequencyValue: parseInt(e.target.value) || 1,
                  }))
                }
                min="1"
                className="w-20"
              />
              <span className="text-sm font-medium">
                {formData.frequency
                  ? formData.frequency.replace(/ly$/, "")
                  : ""}
              </span>
            </div>
          </div>
          {showTimeRange && (
            <div className="mt-4">
              <div className="flex justify-between items-center mb-2">
                <label className="block text-sm font-medium">
                  Time Ranges
                </label>
                <Button
                  type="button"
                  variant="outline"
                  size="sm"
                  onClick={addTimeRange}
                >
                  <Plus className="h-4 w-4 mr-2" />
                  Add Time Range
                </Button>
              </div>
              {formData.timeRanges.map((range) => (
                <div
                  key={range.id}
                  className="flex gap-2 mb-2 items-center"
                >
                  <TimePicker
                    value={range.startTime}
                    onChange={(value) =>
                      updateTimeRange(range.id, "startTime", value)
                    }
                    label="Start Time"
                  />
                  <span>to</span>
                  <TimePicker
                    value={range.endTime}
                    onChange={(value) =>
                      updateTimeRange(range.id, "endTime", value)
                    }
                    label="End Time"
                  />
                  <Button
                    type="button"
                    variant="ghost"
                    size="sm"
                    onClick={() => removeTimeRange(range.id)}
                  >
                    <X className="h-4 w-4" />
                  </Button>
                </div>
              ))}
            </div>
          )}

          <DialogFooter>
            <Button type="button" variant="outline" onClick={onClose}>
              Cancel
            </Button>
            <Button
              type="submit"
              disabled={formData.targetId.length === 0}
            >
              Create Rule
            </Button>
          </DialogFooter>
        </form>
      </DialogContent>
    </Dialog>
  );
};

CreateRuleForm.propTypes = {
  isOpen: PropTypes.bool.isRequired,
  onClose: PropTypes.func.isRequired,
  onSubmit: PropTypes.func.isRequired,
};

export default CreateRuleForm;

export const RulesList = ({ rules = [], onEdit, onDelete }) => {
  // [Previous state declarations remain the same]
  const [editingRule, setEditingRule] = useState(null);
  const [deleteConfirm, setDeleteConfirm] = useState(null);
  const [errorDialog, setErrorDialog] = useState(null);
  const [searchTerm, setSearchTerm] = useState("");
  const [searchResults, setSearchResults] = useState([]);
  const { changeRuleStatus } = useRules();
  const [activeTargetLevel, setActiveTargetLevel] =
    useState("campaign");
  const [loading, setLoading] = useState(false);

  const handleTargetSearch = async (searchTerm) => {
    setSearchTerm(searchTerm);
    if (searchTerm.length < 2) {
      setSearchResults([]);
      return;
    }

    setLoading(true);
    try {
      let endpoint;
      switch (editingRule.targetLevel) {
        case "campaign":
          endpoint = "/auth/db/campaigns";
          break;
        case "adSet":
          endpoint = "/auth/db/ads_set";
          break;
        case "ad":
          endpoint = "/auth/db/ads";
          break;
      }

      const response = await api.get(endpoint);
      const filteredResults = response.data.filter((item) => {
        const searchValue = searchTerm.toLowerCase();
        switch (editingRule.targetLevel) {
          case "campaign":
            return item.campaignName
              .toLowerCase()
              .includes(searchValue);
          case "adSet":
            return item.adSetName.toLowerCase().includes(searchValue);
          case "ad":
            return item.adName.toLowerCase().includes(searchValue);
          default:
            return false;
        }
      });

      setSearchResults(filteredResults);
    } catch (error) {
      console.error("Search error:", error);
      setSearchResults([]);
    } finally {
      setLoading(false);
    }
  };
  const StatusToggleButton = ({ rule, onToggle }) => {
    return (
      <Button
        variant="ghost"
        size="sm"
        onClick={() => onToggle(rule)}
        className="flex items-center gap-2"
      >
        {rule.status === "active" ? (
          <Pause className="h-4 w-4 text-amber-500" />
        ) : (
          <Play className="h-4 w-4 text-emerald-500" />
        )}
        {rule.status === "active" ? "Pause" : "Activate"}
      </Button>
    );
  };
  
  const getStatusIcon = (status, rule) => {
    const iconClass = "h-4 w-4 shrink-0";
    switch (status) {
      case "active":
        return (
          <CheckCircle2 className={`${iconClass} text-emerald-500`} />
        );
      case "paused":
        return (
          <PauseCircle className={`${iconClass} text-amber-500`} />
        );
      case "error":
        return (
          <button
            onClick={(e) => {
              e.stopPropagation();
              setErrorDialog(rule);
            }}
            className="inline-flex items-center gap-1 text-red-500 hover:text-red-600"
          >
            <AlertCircle className={iconClass} />
            <AlertTriangle
              className={`${iconClass} ml-1 cursor-pointer`}
            />
          </button>
        );
      default:
        return null;
    }
  };
  useEffect(() => {
    if (editingRule?.targetId?.length > 0) {
      const fetchInitialTargets = async () => {
        setLoading(true);
        try {
          let endpoint;
          switch (editingRule.targetLevel) {
            case "campaign":
              endpoint = `/auth/db/campaigns?ids=${editingRule.targetId.join(
                ","
              )}`;
              break;
            case "adSet":
              endpoint = `/auth/db/ads_set?ids=${editingRule.targetId.join(
                ","
              )}`;
              break;
            case "ad":
              endpoint = `/auth/db/ads?ids=${editingRule.targetId.join(
                ","
              )}`;
              break;
          }

          const response = await api.get(endpoint);
          setSearchResults(response.data);
        } catch (error) {
          console.error("Error fetching initial targets:", error);
        } finally {
          setLoading(false);
        }
      };

      fetchInitialTargets();
    }
  }, [editingRule?.targetId, editingRule?.targetLevel]);

  const handleStatusToggle = async (rule) => {
    const newStatus = rule.status === "active" ? "paused" : "active";

    try {
      await changeRuleStatus(rule._id, newStatus);
      toast.success(
        `Rule ${
          newStatus === "active" ? "activated" : "paused"
        } successfully`
      );
    } catch (error) {
      console.error("Status toggle error:", error);
      ErrorToast(
        `Failed to ${
          newStatus === "active" ? "activate" : "pause"
        } rule`
      );
    }
  };
  // Validate rule before rendering
  const isValidRule = (rule) => {
    return (
      rule &&
      rule._id &&
      rule.name &&
      rule.targetLevel &&
      Array.isArray(rule.conditions) &&
      Array.isArray(rule.actions) &&
      rule.frequency &&
      rule.status
    );
  };

  // Filter out invalid rules
  const validRules = rules.filter(isValidRule);

  const handleEditClick = (rule) => {
    setEditingRule(rule);
  };

  const handleEditSubmit = async (updatedData) => {
    if (!editingRule?._id) return;
    console.log("updatedData", updatedData);
    try {
      await onEdit(editingRule._id, updatedData);
      toast.success(`Rule updated sucessfully`);
      setEditingRule(null);
    } catch (error) {
      setEditingRule((prev) => ({
        ...prev,
        status: "error",
        errorMessage: error.message || "Failed to update rule",
        lastErrorTime: new Date().toISOString(),
      }));
      ErrorToast(`Error updating rule`);
      console.error("Error updating rule:", error);
    }
  };

  const addTimeRange = () => {
    const now = new Date();
    const start = new Date(now.setHours(9, 0, 0));
    const end = new Date(now.setHours(17, 0, 0));

    setEditingRule((prev) => ({
      ...prev,
      timeRanges: [
        ...(prev.timeRanges || []),
        {
          id: crypto.randomUUID(),
          startTime: start.toISOString(),
          endTime: end.toISOString(),
        },
      ],
    }));
  };

  const updateTimeRange = (id, field, value) => {
    setEditingRule((prev) => ({
      ...prev,
      timeRanges: prev.timeRanges.map((range) =>
        range.id === id ? { ...range, [field]: value } : range
      ),
    }));
  };

  const removeTimeRange = (id) => {
    setEditingRule((prev) => ({
      ...prev,
      timeRanges: prev.timeRanges.filter((range) => range.id !== id),
    }));
  };

  return (
    <Card className="shadow-sm bg-white rounded-lg border border-gray-200">
      <CardContent className="p-6">
        {!validRules?.length ? (
          <Alert className="bg-gray-50 border-gray-200">
            <AlertDescription className={TYPOGRAPHY.body}>
              No rules found. Create a rule to get started.
            </AlertDescription>
          </Alert>
        ) : (
          <div className="relative overflow-hidden rounded-lg">
            <Table className="min-w-full divide-y divide-gray-200">
              <TableHeader className={TABLE_STYLES.header}>
                <TableRow>
                  <TableHead
                    className={`${TABLE_STYLES.headerCell} hidden md:table-cell`}
                  >
                    Rule Name
                  </TableHead>
                  <TableHead className="hidden lg:table-cell">
                    Target Level
                  </TableHead>
                  <TableHead className="hidden xl:table-cell">
                    Conditions
                  </TableHead>
                  <TableHead className="hidden xl:table-cell">
                    Actions
                  </TableHead>
                  <TableHead className="hidden lg:table-cell">
                    Frequency
                  </TableHead>
                  <TableHead>Status</TableHead>
                  {/* <TableHead>Status</TableHead> */}
                  <TableHead className="text-right">
                    Actions
                  </TableHead>
                </TableRow>
              </TableHeader>

              <TableBody className="bg-white divide-y divide-gray-200">
                {validRules.map((rule) => (
                  <TableRow key={rule._id} className={TABLE_STYLES.row}>
                  {/* Name Column */}
                  <TableCell className={`${TABLE_STYLES.cell} font-medium`}>
                    <div className={TYPOGRAPHY.body}>{rule.name}</div>
                  </TableCell>
                
                  {/* Target Level Column */}
                  <TableCell className="hidden lg:table-cell">
                    {rule.targetLevel}
                  </TableCell>
                
                  {/* Conditions Column */}
                  <TableCell className="hidden xl:table-cell">
                    {rule.conditions?.length || 0} conditions
                  </TableCell>
                
                  {/* Actions Column */}
                  <TableCell className="hidden xl:table-cell">
                    {rule.actions?.length || 0} actions
                  </TableCell>
                
                  {/* Frequency Column */}
                  <TableCell className="hidden lg:table-cell">
                    {rule.frequency}
                  </TableCell>
                
                  {/* Status Column */}
                  <TableCell>
                    <span className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium ${BADGE_STYLES[rule.status]}`}>
                      {getStatusIcon(rule.status, rule)}
                      <span className="ml-1">{rule.status}</span>
                    </span>
                  </TableCell>
                
                  {/* Actions Column */}
                  <TableCell>
                    <div className="flex justify-end items-center space-x-2">
                      <Button
                        variant="ghost"
                        size="sm"
                        onClick={() => handleEditClick(rule)}
                        className="text-gray-700 hover:bg-gray-100"
                      >
                        <Edit2 className="h-4 w-4" />
                      </Button>
                      <StatusToggleButton rule={rule} onToggle={handleStatusToggle} />
                      <Button
                        variant="ghost"
                        size="sm"
                        onClick={() => setDeleteConfirm(rule)}
                        className="text-error-DEFAULT hover:bg-error-light"
                      >
                        <Trash2 className="h-4 w-4" />
                      </Button>
                    </div>
                  </TableCell>
                </TableRow>
                ))}
              </TableBody>
            </Table>
          </div>
        )}

        {/* [Dialog components remain the same but with updated styling] */}
        <Dialog
          open={!!errorDialog}
          onOpenChange={() => setErrorDialog(null)}
        >
          <DialogContent className="sm:max-w-lg">
            <DialogHeader>
              <DialogTitle className="flex items-center gap-2 text-red-600">
                <AlertCircle className="h-5 w-5" />
                Rule Error Details
              </DialogTitle>
            </DialogHeader>
            <div className="space-y-4">
              <div className="p-4 bg-red-50 rounded-lg border border-red-100">
                <h4 className="font-medium mb-2">
                  Rule: {errorDialog?.name}
                </h4>
                <p className="text-red-600">
                  {errorDialog?.errorMessage ||
                    "An error occurred while executing this rule."}
                </p>
              </div>
              <div className="text-sm text-gray-500">
                <p>
                  Last error occurred:{" "}
                  {errorDialog?.lastErrorTime
                    ? new Date(
                        errorDialog.lastErrorTime
                      ).toLocaleString()
                    : "Unknown"}
                </p>
              </div>
            </div>
            <DialogFooter>
              <Button onClick={() => setErrorDialog(null)}>
                Close
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>

        {/* EDITTING RULE SECTION */}

        <Dialog
          open={!!editingRule}
          onOpenChange={() => setEditingRule(null)}
        >
          <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
            <DialogHeader>
              <DialogTitle>
                Edit Rule: {editingRule?.name}
              </DialogTitle>
            </DialogHeader>

            <form
              onSubmit={(e) => {
                e.preventDefault();
                handleEditSubmit(editingRule);
              }}
              className="space-y-6"
            >
              {/* Rule Name */}
              <div>
                <label className="block text-sm font-medium mb-1">
                  Rule Name
                </label>
                <Input
                  value={editingRule?.name || ""}
                  onChange={(e) =>
                    setEditingRule((prev) => ({
                      ...prev,
                      name: e.target.value,
                    }))
                  }
                  placeholder="Enter rule name"
                  required
                />
              </div>
              {!searchTerm && searchResults.length > 0 && (
                <div className="space-y-2 max-h-60 overflow-y-auto border rounded-md p-2">
                  <div className="text-sm font-medium p-2 border-b">
                    Selected {editingRule?.targetLevel}s
                  </div>
                  {searchResults.map((result) => {
                    const name =
                      editingRule?.targetLevel === "campaign"
                        ? result.campaignName
                        : editingRule?.targetLevel === "adSet"
                        ? result.adSetName
                        : result.adName;

                    const id =
                      editingRule?.targetLevel === "campaign"
                        ? result.campaignId
                        : editingRule?.targetLevel === "adSet"
                        ? result.adSetId
                        : result.adId;

                    return (
                      <div
                        key={id}
                        className="flex items-center gap-2 p-2 hover:bg-gray-50"
                      >
                        <label
                          htmlFor={id}
                          className="flex-1 cursor-pointer"
                        >
                          <div className="text-sm font-medium">
                            {name}
                          </div>
                          <div className="text-xs text-gray-500">
                            ID: {id}
                          </div>
                        </label>
                      </div>
                    );
                  })}
                </div>
              )}

              {/* Target Selection */}
              <div className="space-y-4">
                <div className="flex space-x-2 border-b">
                  {[
                    { id: "campaign", label: "Campaigns" },
                    { id: "adSet", label: "Ad Sets" },
                    { id: "ad", label: "Ads" },
                  ].map((tab) => (
                    <button
                      key={tab.id}
                      type="button"
                      className={cn(
                        "px-4 py-2 focus:outline-none transition-colors",
                        editingRule?.targetLevel === tab.id
                          ? "border-b-2 border-blue-500 text-blue-600"
                          : "text-gray-500 hover:text-gray-700"
                      )}
                      onClick={(e) => {
                        e.preventDefault();
                        setActiveTargetLevel(tab.id);
                        setEditingRule((prev) => ({
                          ...prev,
                          targetLevel: tab.id,
                          targetId: [],
                        }));
                        setSearchTerm("");
                        setSearchResults([]);
                      }}
                    >
                      {tab.label}
                    </button>
                  ))}
                </div>

                <div className="space-y-4">
                  <div className="relative">
                    <Input
                      type="text"
                      placeholder={`Search ${activeTargetLevel}s...`}
                      value={searchTerm}
                      onChange={(e) =>
                        handleTargetSearch(e.target.value)
                      }
                      className="pr-10"
                    />
                    {loading ? (
                      <div className="absolute right-3 top-1/2 transform -translate-y-1/2">
                        <div className="animate-spin h-4 w-4 border-2 border-blue-500 rounded-full border-t-transparent" />
                      </div>
                    ) : (
                      <Search className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400" />
                    )}
                  </div>

                  {searchResults.length > 0 && (
                    <div className="space-y-2 max-h-60 overflow-y-auto border rounded-md p-2">
                      <div className="flex items-center gap-2 p-2 border-b">
                        <input
                          type="checkbox"
                          id="select-all"
                          checked={
                            editingRule?.targetId?.length ===
                            searchResults.length
                          }
                          onChange={(e) => {
                            const newTargetId = e.target.checked
                              ? searchResults.map((item) =>
                                  editingRule.targetLevel ===
                                  "campaign"
                                    ? item.campaignId
                                    : editingRule.targetLevel ===
                                      "adSet"
                                    ? item.adSetId
                                    : item.adId
                                )
                              : [];

                            setEditingRule((prev) => ({
                              ...prev,
                              targetId: newTargetId,
                            }));
                          }}
                          className="rounded border-gray-300"
                        />

                        <label
                          htmlFor="select-all"
                          className="font-medium"
                        >
                          Select All Results
                        </label>
                      </div>

                      {searchResults.map((result) => {
                        const name =
                          editingRule?.targetLevel === "campaign"
                            ? result.campaignName
                            : editingRule?.targetLevel === "adSet"
                            ? result.adSetName
                            : result.adName;

                        // Correctly select the appropriate ID based on target level
                        const id =
                          editingRule?.targetLevel === "campaign"
                            ? result.campaignId
                            : editingRule?.targetLevel === "adSet"
                            ? result.adSetId
                            : result.adId;

                        return (
                          <div
                            key={id}
                            className="flex items-center gap-2 p-2 hover:bg-gray-50"
                          >
                            <input
                              type="checkbox"
                              id={id}
                              checked={editingRule?.targetId?.includes(
                                id
                              )}
                              onChange={() => {
                                setEditingRule((prev) => ({
                                  ...prev,
                                  targetId: prev.targetId?.includes(
                                    id
                                  )
                                    ? prev.targetId.filter(
                                        (targetId) => targetId !== id
                                      )
                                    : [...(prev.targetId || []), id],
                                }));
                              }}
                              className="rounded border-gray-300"
                            />
                            <label
                              htmlFor={id}
                              className="flex-1 cursor-pointer"
                            >
                              <div className="text-sm font-medium">
                                {name}
                              </div>
                              <div className="text-xs text-gray-500">
                                ID: {id}
                              </div>
                            </label>
                          </div>
                        );
                      })}
                    </div>
                  )}

                  {searchTerm.length >= 2 &&
                    searchResults.length === 0 &&
                    !loading && (
                      <p className="text-sm text-gray-500 p-2">
                        No {editingRule?.targetLevel}s found matching
                        your search.
                      </p>
                    )}
                </div>
              </div>

              {/* Conditions */}
              <div>
                <div className="flex justify-between items-center mb-2">
                  <label className="block text-sm font-medium">
                    Conditions
                  </label>
                  <Button
                    type="button"
                    variant="outline"
                    size="sm"
                    onClick={() => {
                      const now = new Date();
                      const start = new Date(now.setHours(9, 0, 0));
                      const end = new Date(now.setHours(17, 0, 0));

                      const newCondition = {
                        id: crypto.randomUUID(),
                        metric: "",
                        operator: "",
                        value: "",
                        timeframe: "",
                        timeRanges: [{ start, end }],
                        logicalOperator:
                          editingRule?.conditions?.length > 0
                            ? "AND"
                            : "",
                      };
                      setEditingRule((prev) => ({
                        ...prev,
                        conditions: [
                          ...(prev.conditions || []),
                          newCondition,
                        ],
                      }));
                    }}
                  >
                    <Plus className="h-4 w-4 mr-2" />
                    Add Condition
                  </Button>
                </div>
                {editingRule?.conditions?.map((condition, index) => (
                  <div
                    key={condition.id || index}
                    className="space-y-2 mb-4"
                  >
                    {index > 0 && (
                      <Select
                        value={condition.logicalOperator}
                        onValueChange={(value) => {
                          setEditingRule((prev) => ({
                            ...prev,
                            conditions: prev.conditions.map((c) =>
                              c.id === condition.id
                                ? { ...c, logicalOperator: value }
                                : c
                            ),
                          }));
                        }}
                      >
                        <SelectTrigger className="w-20">
                          <SelectValue />
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="AND">AND</SelectItem>
                          <SelectItem value="OR">OR</SelectItem>
                        </SelectContent>
                      </Select>
                    )}
                    <div className="flex gap-2 items-center">
                      <Select
                        value={condition.metric}
                        onValueChange={(value) => {
                          setEditingRule((prev) => ({
                            ...prev,
                            conditions: prev.conditions.map((c) =>
                              c.id === condition.id
                                ? { ...c, metric: value }
                                : c
                            ),
                          }));
                        }}
                      >
                        <SelectTrigger className="w-1/3">
                          <SelectValue placeholder="Select metric" />
                        </SelectTrigger>
                        <SelectContent>
                          {Object.entries(
                            conditionConfig.metrics
                          ).map(([key, config]) => (
                            <SelectItem key={key} value={key}>
                              {config.label}
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>

                      {condition.metric && (
                        <Select
                          value={condition.operator}
                          onValueChange={(value) => {
                            setEditingRule((prev) => ({
                              ...prev,
                              conditions: prev.conditions.map((c) =>
                                c.id === condition.id
                                  ? { ...c, operator: value }
                                  : c
                              ),
                            }));
                          }}
                        >
                          <SelectTrigger className="w-1/3">
                            <SelectValue placeholder="Select operator" />
                          </SelectTrigger>
                          <SelectContent>
                            {conditionConfig?.metrics[
                              condition?.metric
                            ]?.operators?.map((op) => (
                              <SelectItem
                                key={op}
                                value={op || "default"}
                              >
                                {" "}
                                {conditionConfig.operators[op]}
                              </SelectItem>
                            ))}
                          </SelectContent>
                        </Select>
                      )}

                      {condition.operator && (
                        <Input
                          type="number"
                          value={condition.value}
                          onChange={(e) => {
                            setEditingRule((prev) => ({
                              ...prev,
                              conditions: prev.conditions.map((c) =>
                                c.id === condition.id
                                  ? { ...c, value: e.target.value }
                                  : c
                              ),
                            }));
                          }}
                          placeholder="Value"
                          className="w-1/4"
                        />
                      )}

                      <Button
                        type="button"
                        variant="ghost"
                        size="sm"
                        onClick={() => {
                          setEditingRule((prev) => ({
                            ...prev,
                            conditions: prev.conditions.filter(
                              (c) => c.id !== condition.id
                            ),
                          }));
                        }}
                      >
                        <X className="h-4 w-4" />
                      </Button>
                    </div>
                  </div>
                ))}
              </div>

              {/* Actions */}
              <div>
                <div className="flex justify-between items-center mb-2">
                  <label className="block text-sm font-medium">
                    Actions
                  </label>
                  <Button
                    type="button"
                    variant="outline"
                    size="sm"
                    onClick={() => {
                      const newAction = {
                        id: crypto.randomUUID(),
                        type: "",
                        value: "",
                        customValue: "",
                      };
                      setEditingRule((prev) => ({
                        ...prev,
                        actions: [...(prev.actions || []), newAction],
                      }));
                    }}
                  >
                    <Plus className="h-4 w-4 mr-2" />
                    Add Action
                  </Button>
                </div>
                {editingRule?.actions?.map((action, index) => (
                  <div
                    key={action.id || index}
                    className="flex gap-2 mb-2 items-center"
                  >
                    <Select
                      value={action.type}
                      onValueChange={(value) => {
                        setEditingRule((prev) => ({
                          ...prev,
                          actions: prev.actions.map((a) =>
                            a.id === action.id
                              ? { ...a, type: value, value: "" }
                              : a
                          ),
                        }));
                      }}
                    >
                      <SelectTrigger className="w-1/3">
                        <SelectValue placeholder="Select action type" />
                      </SelectTrigger>
                      <SelectContent>
                        <SelectItem value="change_budget">
                          Change Budget
                        </SelectItem>
                        <SelectItem value="change_status">
                          Change Status
                        </SelectItem>
                        <SelectItem value="change_name">
                          Change Name
                        </SelectItem>
                        <SelectItem value="change_bid">
                          Change Bid
                        </SelectItem>
                        <SelectItem value="send_notification">
                          Send Notification
                        </SelectItem>
                      </SelectContent>
                    </Select>

                    {action.type && (
                      <Select
                        value={action.value}
                        onValueChange={(value) => {
                          setEditingRule((prev) => ({
                            ...prev,
                            actions: prev.actions.map((a) =>
                              a.id === action.id ? { ...a, value } : a
                            ),
                          }));
                        }}
                      >
                        <SelectTrigger className="w-1/3">
                          <SelectValue placeholder="Select value" />
                        </SelectTrigger>
                        <SelectContent>
                          {action.type === "change_budget" && (
                            <>
                              <SelectItem value="increase_percent">
                                Increase by %
                              </SelectItem>
                              <SelectItem value="decrease_percent">
                                Decrease by %
                              </SelectItem>
                              <SelectItem value="increase_fixed">
                                Increase by fixed amount
                              </SelectItem>
                              <SelectItem value="decrease_fixed">
                                Decrease by fixed amount
                              </SelectItem>
                              <SelectItem value="set_fixed">
                                Set to fixed amount
                              </SelectItem>
                            </>
                          )}
                          {action.type === "change_status" && (
                            <>
                              <SelectItem value="active">
                                Set Active
                              </SelectItem>
                              <SelectItem value="paused">
                                Set Paused
                              </SelectItem>
                            </>
                          )}
                          {action.type === "change_name" && (
                            <>
                              <SelectItem value="prefix">
                                Add prefix
                              </SelectItem>
                              <SelectItem value="suffix">
                                Add suffix
                              </SelectItem>
                            </>
                          )}
                        </SelectContent>
                      </Select>
                    )}

                    {(action.type === "change_budget" ||
                      action.type === "change_name") && (
                      <Input
                        type={
                          action.type === "change_budget"
                            ? "number"
                            : "text"
                        }
                        value={action.customValue || ""}
                        onChange={(e) => {
                          setEditingRule((prev) => ({
                            ...prev,
                            actions: prev.actions.map((a) =>
                              a.id === action.id
                                ? {
                                    ...a,
                                    customValue: e.target.value,
                                  }
                                : a
                            ),
                          }));
                        }}
                        placeholder={
                          action.type === "change_budget"
                            ? "Enter amount"
                            : "Enter text"
                        }
                        className="w-1/4"
                      />
                    )}

                    <Button
                      type="button"
                      variant="ghost"
                      size="sm"
                      onClick={() => {
                        setEditingRule((prev) => ({
                          ...prev,
                          actions: prev.actions.filter(
                            (a) => a.id !== action.id
                          ),
                        }));
                      }}
                    >
                      <X className="h-4 w-4" />
                    </Button>
                  </div>
                ))}
              </div>

              {/* Frequency */}
              <div className="space-y-4">
                <div>
                  <label className="block text-sm font-medium mb-1">
                    Frequency
                  </label>
                  <div className="flex gap-4">
                    <Select
                      value={editingRule?.frequency}
                      onValueChange={(value) => {
                        setEditingRule((prev) => ({
                          ...prev,
                          frequency: value,
                        }));
                      }}
                    >
                      <SelectTrigger className="w-1/2">
                        <SelectValue placeholder="Select frequency" />
                      </SelectTrigger>
                      <SelectContent>
                        <SelectItem value="minutes">
                          Minutes
                        </SelectItem>
                        <SelectItem value="hourly">Hourly</SelectItem>
                        <SelectItem value="daily">Daily</SelectItem>
                        <SelectItem value="weekly">Weekly</SelectItem>
                      </SelectContent>
                    </Select>
                    <Input
                      type="number"
                      value={editingRule?.frequencyValue || 1}
                      onChange={(e) => {
                        setEditingRule((prev) => ({
                          ...prev,
                          frequencyValue:
                            parseInt(e.target.value) || 1,
                        }));
                      }}
                      min="1"
                      className="w-1/4"
                    />
                  </div>
                </div>

                {/* Time Ranges */}
                {["hourly", "daily", "weekly"].includes(
                  editingRule?.frequency
                ) && (
                  <div>
                    <div className="flex justify-between items-center mb-2">
                      <label className="block text-sm font-medium">
                        Time Ranges
                      </label>
                      <Button
                        type="button"
                        variant="outline"
                        size="sm"
                        onClick={addTimeRange}
                      >
                        <Plus className="h-4 w-4 mr-2" />
                        Add Time Range
                      </Button>
                    </div>
                    {editingRule?.timeRanges?.map((range) => (
                      <div
                        key={range.id}
                        className="flex gap-2 mb-2 items-center"
                      >
                        <TimePicker
                          value={range.startTime}
                          onChange={(value) =>
                            updateTimeRange(
                              range.id,
                              "startTime",
                              value
                            )
                          }
                          label="Start Time"
                        />
                        <span>to</span>
                        <TimePicker
                          value={range.endTime}
                          onChange={(value) =>
                            updateTimeRange(
                              range.id,
                              "endTime",
                              value
                            )
                          }
                          label="End Time"
                        />
                        <Button
                          type="button"
                          variant="ghost"
                          size="sm"
                          onClick={() => removeTimeRange(range.id)}
                        >
                          <X className="h-4 w-4" />
                        </Button>
                      </div>
                    ))}
                  </div>
                )}
              </div>

              <DialogFooter>
                <Button
                  type="button"
                  variant="outline"
                  onClick={() => setEditingRule(null)}
                >
                  Cancel
                </Button>
                <Button type="submit">Save Changes</Button>
              </DialogFooter>
            </form>
          </DialogContent>
        </Dialog>

        {/* CONFIRMATION SECTION */}
        <Dialog
          open={!!deleteConfirm}
          onOpenChange={() => setDeleteConfirm(null)}
        >
          <DialogContent className="sm:max-w-lg">
            <DialogHeader>
              <DialogTitle>Delete Rule</DialogTitle>
            </DialogHeader>
            <p className="text-gray-600">
              Are you sure you want to delete the rule "
              {deleteConfirm?.name}"? This action cannot be undone.
            </p>
            <DialogFooter>
              <Button
                variant="outline"
                onClick={() => setDeleteConfirm(null)}
              >
                Cancel
              </Button>
              <Button
                variant="destructive"
                onClick={() => {
                  if (deleteConfirm?._id) {
                    onDelete(deleteConfirm._id);
                  }
                  setDeleteConfirm(null);
                }}
              >
                Delete Rule
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>
      </CardContent>
    </Card>
  );
};

// [PropTypes remain the same]
RulesList.propTypes = {
  rules: PropTypes.arrayOf(
    PropTypes.shape({
      _id: PropTypes.string.isRequired,
      name: PropTypes.string.isRequired,
      targetLevel: PropTypes.string.isRequired,
      conditions: PropTypes.arrayOf(
        PropTypes.shape({
          metric: PropTypes.string.isRequired,
          operator: PropTypes.string.isRequired,
          value: PropTypes.oneOfType([
            PropTypes.string,
            PropTypes.number,
          ]).isRequired,
        })
      ).isRequired,
      actions: PropTypes.arrayOf(
        PropTypes.shape({
          type: PropTypes.string.isRequired,
          value: PropTypes.oneOfType([
            PropTypes.string,
            PropTypes.number,
          ]).isRequired,
        })
      ).isRequired,
      frequency: PropTypes.string.isRequired,
      status: PropTypes.string.isRequired,
      errorMessage: PropTypes.string,
      lastErrorTime: PropTypes.string,
    })
  ),
  onEdit: PropTypes.func.isRequired,
  onDelete: PropTypes.func.isRequired,
};

export default RulesList;
